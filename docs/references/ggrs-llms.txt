## Config (compile-time parameterization)
pub trait Config: 'static {
    type Input: Copy + Clone + PartialEq + Default + Serialize + DeserializeOwned;
    type InputPredictor: InputPredictor<Self::Input>;   // PredictRepeatLast is the documented reasonable default
    type State;
    type Address: Clone + PartialEq + Eq + Hash + Debug;
}
# Input::default() is what GGRS feeds for disconnected players ("no input").

## Transport
pub trait NonBlockingSocket<A> where A: Clone + PartialEq + Eq + Hash {
    fn send_to(&mut self, msg: &Message, addr: &A);
    fn receive_all_messages(&mut self) -> Vec<(A, Message)>;
}
# Expected: discrete, UNORDERED, UNRELIABLE packets. GGRS adds its own reliability/ordering.
# -> Do NOT put GGRS on an ordered+reliable WebRTC channel (head-of-line blocking).
# Message serialization for our wire format: postcard via serde (verify Message: Serialize + Deserialize in Phase 1 spike;
#   if not public-serde, fall back to the crate's own encoding helpers).

## Requests — MUST be handled in order
pub enum GgrsRequest<T: Config> {
    SaveGameState { cell: GameStateCell<T::State>, frame: Frame },
    LoadGameState { cell: GameStateCell<T::State>, frame: Frame },
    AdvanceFrame  { inputs: Vec<(T::Input, InputStatus)> },   // InputStatus: Confirmed | Predicted | Disconnected
}
GameStateCell<T>:
  pub fn save(&self, frame: Frame, data: Option<T>, checksum: Option<u128>)
  pub fn load(&self) -> Option<T>            where T: Clone
  pub fn data(&self) -> Option<GameStateAccessor<'_, T>>   // no-clone access, niche

## Events
pub enum GgrsEvent<T: Config> {
    Synchronizing { addr, total: u32, count: u32 },
    Synchronized { addr },
    Disconnected { addr },
    NetworkInterrupted { addr, disconnect_timeout: u128 },
    NetworkResumed { addr },
    WaitRecommendation { skip_frames: u32 },
    DesyncDetected { frame: Frame, local_checksum: u128, remote_checksum: u128, addr },
}

## Our mapping (see design-docs/netcode-rollback.md)
Input=PlayerInput(u16) · Predictor=PredictRepeatLast · State=sim::State (Copy) · Address=PeerSlot(u8)
checksum = xxh3_64(bytemuck::bytes_of(&state)) as u128
